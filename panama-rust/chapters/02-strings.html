isMatch(pat, hay) — JNI

static native boolean isMatch(String pat, String hay);
boolean m = JniRsRegex.isMatch("café", "un café");
#[no_mangle]
pub extern "system"
fn Java_org_example_rust_JniRsRegex_isMatch(
    mut env: JNIEnv,
    _class: JClass,
    pattern: JString, // pointer to Java String
    haystack: JString, // pointer to Java String
) -> jboolean {
    let pat: String = env.get_string(&pattern)
      .unwrap().into();
    let hay: String = env.get_string(&haystack)
      .unwrap().into();
    Regex::new(&pat).unwrap().is_match(&hay)
      as jboolean
}
  • JNI function name is mangled: Java_org_example_rust_JniRsRegex_isMatch
  • Java side looks free: a plain static call. Rust gets a JString: a reference to the Java object, no bytes
  • "café" in the JVM is LATIN1: é is one byte, E9
  • env.get_string: the JVM copies it into a C buffer as modified UTF-8 (é → C3 A9, NUL-terminated)
  • .into(): Rust copies again into its own UTF-8 String
  • 2 copies + 2 conversions, both inside Rust, through JNI calls: hidden from the Java code

isMatch(pat, hay) — Panama

static final MethodHandle IS_MATCH = Linker
          .libraryLookup(...)
          .downcallHandle(...);

public boolean isMatch(String pat, String hay) {
    try (Arena arena = Arena.ofConfined()) {
        MemorySegment cP = arena.allocateFrom(
            JAVA_BYTE,
            pat.getBytes(UTF_8));
        MemorySegment cH = arena.allocateFrom(
            JAVA_BYTE,
            hay.getBytes(UTF_8));
        return (boolean) IS_MATCH.invoke(
            cP, cP.byteSize(), cH, cH.byteSize());
    }
}
// Rust: a view on the arena bytes, no copy
let pat = unsafe { std::str::from_utf8_unchecked(
    std::slice::from_raw_parts(pat_ptr, pat_len)) };
  • Rust side: plain is_match_once(ptr, len, ptr, len), a view on the arena bytes with from_raw_parts: no copy in Rust
  • getBytes(UTF_8): Java converts LATIN1 → UTF-8 (é: E9 → C3 A9) into a heap byte[]
  • allocateFrom: copies those 5 bytes off-heap into the arena
  • still 2 copies, but in Java and in plain sight: the JIT can optimize them
  • pass cP.byteSize() = 5 bytes, not pat.length() = 4 chars: with "café" they differ
  • leaving the try: the arena closes and frees its bytes; the heap byte[] is now garbage, the GC collects it whenever it wants
