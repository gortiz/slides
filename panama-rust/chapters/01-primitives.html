Panama

#[no_mangle]
pub extern "C"
fn add(a: i32, b: i32) -> i32 {
    a + b
}
public class Lib {
    static { System.loadLibrary("mylib"); }
    public static native int add(int a, int b);
}

int r = Lib.add(2, 3);   // → 5
private static final MethodHandle ADD;
static {
    SymbolLookup lib = SymbolLookup.libraryLookup(
        "libmylib.so", Arena.global());
    ADD = Linker.nativeLinker().downcallHandle(
        lib.find("add").orElseThrow(),
        FunctionDescriptor.of(
            JAVA_INT, JAVA_INT, JAVA_INT));
}
int r = (int) ADD.invoke(2, 3);   // → 5
  • same function, both bridges — top row Rust, bottom Java; JNI left, Panama right
  • Panama (focus): plain extern "C" add; Java wires it with libraryLookup + downcallHandle + a FunctionDescriptor, then ADD.invoke
  • JNI (quick contrast): mangled name, injected JNIEnv/JClass, but a plain static call
  • verdict: for trivial cases JNI wins on ergonomics — Panama's boilerplate is visible. Payoff comes when the function does something real
